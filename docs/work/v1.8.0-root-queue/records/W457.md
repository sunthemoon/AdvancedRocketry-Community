The scratch CLI check was run as this PowerShell here-string piped to `python -B -`. The helper creates the same
disposable, platform-neutral repository fixture used by the existing validator tests; only its ledger is edited.

```python
import csv
import io
import re
import subprocess
import sys
from pathlib import Path

sys.path.insert(0, str(Path('tests').resolve()))
from test_validate_v180_content_ledger import V180ContentLedgerTests
from scripts.validate_v180_content_ledger import LEDGER

source = Path(LEDGER).read_bytes()
draft = Path('docs/work/v1.8.0-root-queue/records/ADR-062-REVISION-8-DRAFT.md').read_text(encoding='utf-8')
blocks = re.findall(r'^```csv\n(.*?)\n```$', draft, re.M | re.S)
assert len(blocks) == 1
proposed = list(csv.DictReader(io.StringIO(blocks[0])))
assert len(proposed) == 5
replacements = {row['unit_id']: row for row in proposed}
assert set(replacements) == {f'block_variant:loader/{index}' for index in range(1, 6)}
case = V180ContentLedgerTests()
case.setUp()
try:
    rows = case._rows(LEDGER)
    case._write_rows(LEDGER, [replacements.get(row['unit_id'], row) for row in rows])
    subprocess.run([sys.executable, '-B', 'scripts/validate_v180_content_ledger.py',
                    '--root', str(case.root), '--closure'], check=True)
    assert Path(LEDGER).read_bytes() == source
    print('Scratch --closure passed; authoritative ledger bytes unchanged.')
finally:
    case.tearDown()
```
